```sql
-- Top 5 customers by total spending
SELECT
    c.id,
    c.name,
    SUM(o.total) as total_spending
FROM customers c
LEFT JOIN orders o ON c.id = o.customer_id
GROUP BY c.id, c.name
ORDER BY total_spending DESC
LIMIT 5;
```
